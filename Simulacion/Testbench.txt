module view_selector_case#(
parameter integer N=4
)(
input wire clk,
output reg done
);
reg rst_n;
reg key_s2;
reg frame_start;
wire mode_fft;
reg expected_mode;
integer checks;
integer i;
view_selector#(.DEBOUNCE_CYCLES(N))dut(
.clk(clk),
.rst_n(rst_n),
.key_s2(key_s2),
.frame_start(frame_start),
.mode_fft(mode_fft)
);
task automatic check_output;
begin
checks=checks+1;
if(mode_fft!==expected_mode)
$fatal(1,
"N=%0d comprobacion=%0d t=%0t: modo=%b esperado=%b key=%b frame=%b reset=%b",
N,checks,$time,mode_fft,expected_mode,
key_s2,frame_start,rst_n);
end
endtask
task automatic cycle(
input reg next_key,
input reg next_frame,
input reg expected_after
);
begin
@(negedge clk);
key_s2=next_key;
frame_start=next_frame;
expected_mode=expected_after;
@(posedge clk);
#1;
check_output();
end
endtask
task automatic hold_key(
input reg level,
input integer cycles,
input reg expected_after
);
integer j;
begin
for(j=0;j<cycles;j=j+1)
cycle(level,1'b0,expected_after);
end
endtask
task automatic publish(
input reg level,
input reg expected_after
);
begin
cycle(level,1'b1,expected_after);
cycle(level,1'b0,expected_after);
end
endtask
task automatic reset_case(input reg held_key);
begin
@(negedge clk);
rst_n=1'b0;
key_s2=held_key;
frame_start=1'b0;
expected_mode=1'b0;
#1;
check_output();
repeat(2)@(posedge clk);
#1;
rst_n=1'b1;
end
endtask
task automatic unsampled_pulse(
input reg stable_level,
input reg expected_after
);
begin
@(negedge clk);
frame_start=1'b0;
expected_mode=expected_after;
#1key_s2=~stable_level;
#1key_s2=stable_level;
@(posedge clk);
#1;
check_output();
end
endtask
task automatic bounce(
input reg stable_level,
input reg expected_after
);
integer j;
begin
for(j=0;j<4;j=j+1)begin
if(N==1)begin
unsampled_pulse(stable_level,expected_after);
end else begin
cycle(~stable_level,1'b0,expected_after);
cycle(stable_level,1'b0,expected_after);
end
end
hold_key(stable_level,N+2,expected_after);
end
endtask
initial begin
rst_n=1'b0;
key_s2=1'b0;
frame_start=1'b0;
expected_mode=1'b0;
checks=0;
done=1'b0;
reset_case(1'b0);
hold_key(1'b0,N+2,1'b0);
publish(1'b0,1'b0);
if(N>1)begin
hold_key(1'b1,N-1,1'b0);
hold_key(1'b0,N+2,1'b0);
end else begin
unsampled_pulse(1'b0,1'b0);
hold_key(1'b0,N+2,1'b0);
end
publish(1'b0,1'b0);
bounce(1'b0,1'b0);
publish(1'b0,1'b0);
hold_key(1'b1,N+2,1'b0);
publish(1'b1,1'b1);
reset_case(1'b0);
hold_key(1'b1,N+1,1'b0);
cycle(1'b1,1'b1,1'b0);
cycle(1'b1,1'b0,1'b0);
publish(1'b1,1'b1);
for(i=0;i<24;i=i+1)begin
cycle(1'b1,1'b0,1'b1);
publish(1'b1,1'b1);
end
bounce(1'b1,1'b1);
publish(1'b1,1'b1);
if(N>1)begin
hold_key(1'b0,N-1,1'b1);
hold_key(1'b1,N+2,1'b1);
publish(1'b1,1'b1);
end
hold_key(1'b0,N+2,1'b1);
hold_key(1'b1,N+2,1'b1);
publish(1'b1,1'b0);
reset_case(1'b0);
hold_key(1'b1,N+2,1'b0);
hold_key(1'b0,N+2,1'b0);
hold_key(1'b1,N+2,1'b0);
hold_key(1'b0,N+2,1'b0);
publish(1'b0,1'b0);
reset_case(1'b0);
for(i=0;i<3;i=i+1)begin
hold_key(1'b1,N+2,1'b0);
hold_key(1'b0,N+2,1'b0);
end
publish(1'b0,1'b1);
reset_case(1'b1);
hold_key(1'b1,N+1,1'b0);
cycle(1'b1,1'b1,1'b0);
publish(1'b1,1'b1);
hold_key(1'b1,N+4,1'b1);
publish(1'b1,1'b1);
reset_case(1'b0);
hold_key(1'b1,N+2,1'b0);
reset_case(1'b0);
hold_key(1'b0,N+2,1'b0);
publish(1'b0,1'b0);
hold_key(1'b1,N+1,1'b0);
reset_case(1'b0);
hold_key(1'b0,N+2,1'b0);
publish(1'b0,1'b0);
$display("PASS: DEBOUNCE_CYCLES=%0d, comprobaciones=%0d",N,checks);
done=1'b1;
end
endmodule

module tb_sample_capture;
reg clk=1'b0;
always#5clk=~clk;
reg rst_n;
reg sample_valid;
reg signed[15:0]sample_signed;
wire block_valid;
wire block_bank;
wire[31:0]block_seq;
reg block_release;
wire overrun_latched;
reg rd_en;
reg rd_bank;
reg[7:0]rd_addr;
wire signed[15:0]rd_sample;
wire rd_valid;
reg expected_valid;
reg expected_bank;
reg[31:0]expected_seq;
reg expected_overrun;
reg expected_next_bank;
reg[31:0]expected_next_seq;
integer expected_pattern_id;
integer checks;
integer read_checks;
integer i;
integer j;
sample_capture dut(
.clk(clk),.rst_n(rst_n),
.sample_valid(sample_valid),.sample_signed(sample_signed),
.block_valid(block_valid),.block_bank(block_bank),.block_seq(block_seq),
.block_release(block_release),.overrun_latched(overrun_latched),
.rd_en(rd_en),.rd_bank(rd_bank),.rd_addr(rd_addr),
.rd_sample(rd_sample),.rd_valid(rd_valid)
);
function automatic signed[15:0]pattern(
input integer block_id,
input integer address
);
begin
case(address)
0:pattern=16'h8000+block_id;
1:pattern=16'h7fff-block_id;
2:pattern=16'hffff;
3:pattern=16'h0000;
254:pattern=16'h1234^block_id;
255:pattern=16'hfedc^block_id;
default:pattern=(address*73+block_id*997)^16'ha55a;
endcase
end
endfunction
task automatic check_outputs(
input reg expected_rd_valid,
input reg signed[15:0]expected_rd_sample
);
begin
checks=checks+1;
if(block_valid!==expected_valid||
block_bank!==expected_bank||block_seq!==expected_seq||
overrun_latched!==expected_overrun)
$fatal(1,
"Metadata t=%0t: valid=%b/%b banco=%b/%b seq=%0d/%0d overrun=%b/%b",
$time,block_valid,expected_valid,block_bank,expected_bank,
block_seq,expected_seq,overrun_latched,expected_overrun);
if(rd_valid!==expected_rd_valid)
$fatal(1,"rd_valid t=%0t: obtenido=%b esperado=%b",
$time,rd_valid,expected_rd_valid);
if(expected_rd_valid)begin
read_checks=read_checks+1;
if(rd_sample!==expected_rd_sample)
$fatal(1,
"Lectura t=%0t banco=%0d addr=%0d dato=%h esperado=%h",
$time,rd_bank,rd_addr,rd_sample,expected_rd_sample);
end
end
endtask
task automatic cycle(
input reg next_sample_valid,
input reg signed[15:0]next_sample,
input reg next_release,
input reg next_rd_en,
input reg next_rd_bank,
input reg[7:0]next_rd_addr,
input integer publication_id,
input reg expect_fault
);
reg expected_read;
reg signed[15:0]expected_word;
begin
@(negedge clk);
sample_valid=next_sample_valid;
sample_signed=next_sample;
block_release=next_release;
rd_en=next_rd_en;
rd_bank=next_rd_bank;
rd_addr=next_rd_addr;
expected_read=next_rd_en&&expected_valid&&
(next_rd_bank==expected_bank);
expected_word=pattern(expected_pattern_id,next_rd_addr);
if(!expected_overrun)begin
if(expect_fault)begin
expected_overrun=1'b1;
end else begin
if(next_release)
expected_valid=1'b0;
if(publication_id>=0)begin
expected_valid=1'b1;
expected_bank=expected_next_bank;
expected_seq=expected_next_seq;
expected_pattern_id=publication_id;
expected_next_bank=~expected_next_bank;
expected_next_seq=expected_next_seq+32'd1;
end
end
end
@(posedge clk);
#1;
check_outputs(expected_read,expected_word);
end
endtask
task automatic reset_capture;
begin
@(negedge clk);
rst_n=1'b0;
sample_valid=1'b0;
sample_signed=16'sd0;
block_release=1'b0;
rd_en=1'b0;
rd_bank=1'b0;
rd_addr=8'd0;
expected_valid=1'b0;
expected_bank=1'b0;
expected_seq=32'd0;
expected_overrun=1'b0;
expected_next_bank=1'b0;
expected_next_seq=32'd0;
expected_pattern_id=-1;
#1;
check_outputs(1'b0,16'sd0);
repeat(2)@(posedge clk);
#1rst_n=1'b1;
end
endtask
task automatic send_block(
input integer block_id,
input reg add_gaps,
input reg publish_at_end,
input reg fault_at_end
);
integer address;
begin
for(address=0;address<256;address=address+1)begin
if(add_gaps&&((address%13)==0))
cycle(1'b0,16'hdead,1'b0,1'b1,
expected_next_bank,address,-1,1'b0);
cycle(1'b1,pattern(block_id,address),1'b0,1'b0,1'b0,
8'd0,((address==255)&&publish_at_end)?block_id:-1,
(address==255)&&fault_at_end);
end
end
endtask
task automatic read_published_block;
integer address;
begin
for(address=0;address<256;address=address+1)
cycle(1'b0,16'sd0,1'b0,1'b1,expected_bank,
address,-1,1'b0);
end
endtask
initial begin
rst_n=1'b0;
sample_valid=1'b0;
sample_signed=16'sd0;
block_release=1'b0;
rd_en=1'b0;
rd_bank=1'b0;
rd_addr=8'd0;
expected_valid=1'b0;
expected_bank=1'b0;
expected_seq=32'd0;
expected_overrun=1'b0;
expected_next_bank=1'b0;
expected_next_seq=32'd0;
expected_pattern_id=-1;
checks=0;
read_checks=0;
reset_capture();
cycle(1'b0,16'sd0,1'b1,1'b1,1'b0,8'd255,-1,1'b0);
cycle(1'b0,16'sd0,1'b0,1'b1,1'b1,8'd0,-1,1'b0);
send_block(0,1'b1,1'b1,1'b0);
cycle(1'b0,16'sd0,1'b0,1'b1,1'b0,8'd255,-1,1'b0);
cycle(1'b0,16'sd0,1'b0,1'b1,1'b1,8'd0,-1,1'b0);
for(j=1;j<4;j=j+1)begin
for(i=0;i<256;i=i+1)
cycle(1'b1,pattern(j,i),(i==255),1'b1,expected_bank,
(j==2)?(255-i):i,(i==255)?j:-1,1'b0);
end
read_published_block();
cycle(1'b0,16'sd0,1'b1,1'b0,1'b0,8'd0,-1,1'b0);
cycle(1'b0,16'sd0,1'b1,1'b1,expected_bank,8'd255,-1,1'b0);
send_block(4,1'b0,1'b1,1'b0);
read_published_block();
cycle(1'b0,16'sd0,1'b1,1'b1,expected_bank,8'd255,-1,1'b0);
reset_capture();
for(i=0;i<131;i=i+1)
cycle(1'b1,pattern(20,i),1'b0,1'b0,1'b0,8'd0,-1,1'b0);
reset_capture();
cycle(1'b0,16'sd0,1'b0,1'b1,1'b0,8'd130,-1,1'b0);
send_block(21,1'b1,1'b1,1'b0);
read_published_block();
reset_capture();
send_block(10,1'b0,1'b1,1'b0);
send_block(11,1'b0,1'b0,1'b1);
for(i=0;i<600;i=i+1)
cycle(1'b1,pattern(12,i%256),(i==200),1'b1,
1'b0,i%256,-1,1'b0);
cycle(1'b0,16'sd0,1'b1,1'b1,1'b1,8'd0,-1,1'b0);
read_published_block();
reset_capture();
send_block(22,1'b0,1'b1,1'b0);
read_published_block();
cycle(1'b0,16'sd0,1'b1,1'b0,1'b0,8'd0,-1,1'b0);
$display("PASS: sample_capture, comprobaciones=%0d lecturas=%0d",
checks,read_checks);
$finish;
end

module tb_fft_twiddle_rom;
reg clk=1'b0;
always#5clk=~clk;
reg en=1'b0;
reg[6:0]addr=7'd0;
wire signed[17:0]w_re,w_im;
integer index;
integer expected_re,expected_im;
integer previous_re,previous_im;
integer checks=0;
real angle;
localparam real PI=3.14159265358979323846;
fft_twiddle_rom dut(
.clk(clk),.en(en),.addr(addr),.w_re(w_re),.w_im(w_im)
);
function automatic integer nearest_away(input real value);
begin
if(value>=0.0)
nearest_away=$rtoi(value+0.5);
else
nearest_away=-$rtoi(-value+0.5);
end
endfunction
initial begin
repeat(2)@(negedge clk);
for(index=0;index<128;index=index+1)begin
@(negedge clk);
en=1'b1;
addr=index[6:0];
#1;
if(index>0&&
($signed(w_re)!==previous_re||$signed(w_im)!==previous_im))
$fatal(1,"La ROM cambio antes del flanco de lectura: indice %0d",index);
angle=2.0*PI*index/256.0;
expected_re=nearest_away(65536.0*$cos(angle));
expected_im=nearest_away(-65536.0*$sin(angle));
@(posedge clk);
#1;
if($signed(w_re)!==expected_re||$signed(w_im)!==expected_im)
$fatal(1,"W[%0d]=(%0d,%0d), esperado=(%0d,%0d)",
index,$signed(w_re),$signed(w_im),expected_re,expected_im);
checks=checks+1;
previous_re=expected_re;
previous_im=expected_im;
@(negedge clk);
en=1'b0;
addr=7'd127-index[6:0];
repeat(2)begin
@(posedge clk);
#1;
if($signed(w_re)!==previous_re||$signed(w_im)!==previous_im)
$fatal(1,"La ROM no retuvo coeficiente con en=0: indice %0d",index);
checks=checks+1;
end
end
$display("PASS: ROM FFT256, 128 coeficientes y retencion sincrona, %0d comprobaciones",checks);
$finish;
end
initial begin
#100000;
$fatal(1,"Timeout ROM FFT256");
end

`elsif ELDI_TB_MARIPOSA
`timescale 1ns/1ps
module tb_fft_butterfly;
localparam real CLOCK_NS=1000.0/27.0;
reg clk=0;
always#(CLOCK_NS/2.0)clk=!clk;
reg rst_n=0,start=0;
reg signed[17:0]a_re=0,a_im=0,b_re=0,b_im=0,w_re=0,w_im=0;
wire busy,done;
wire signed[17:0]y0_re,y0_im,y1_re,y1_im;
integer cases_checked=0;
reg[31:0]rng=32'h62024a17;
fft_butterfly dut(
.clk(clk),.rst_n(rst_n),.start(start),.a_re(a_re),.a_im(a_im),
.b_re(b_re),.b_im(b_im),.w_re(w_re),.w_im(w_im),
.busy(busy),.done(done),.y0_re(y0_re),.y0_im(y0_im),
.y1_re(y1_re),.y1_im(y1_im)
);
function automatic signed[17:0]round_sat(input longint signed n);
longint signed q;
begin
if(n>=0)q=(n+64'sd65536)/64'sd131072;
else q=-((-n+64'sd65536)/64'sd131072);
if(q>131071)round_sat=18'sd131071;
else if(q<-131072)round_sat=-18'sd131072;
else round_sat=q;
end
endfunction
function automatic signed[17:0]random18;
begin
rng=rng*32'd1664525+32'd1013904223;
random18=$signed(rng[31:14]);
end
endfunction
task automatic reset_dut;
begin
@(negedge clk);rst_n=0;start=0;
repeat(3)@(negedge clk);
rst_n=1;
@(posedge clk);#1;
if(busy||done)$fatal(1,"BF: reset no devuelve reposo");
end
endtask
task automatic run_case(
input logic signed[17:0]ar,ai,br,bi,wr,wi,
input bit inject_busy_start
);
longint signed ar64,ai64,br64,bi64,wr64,wi64,tr,ti;
reg signed[17:0]e0r,e0i,e1r,e1i;
integer elapsed;
begin
ar64=ar;ai64=ai;br64=br;bi64=bi;wr64=wr;wi64=wi;
tr=br64*wr64-bi64*wi64;
ti=br64*wi64+bi64*wr64;
e0r=round_sat(ar64*65536+tr);e0i=round_sat(ai64*65536+ti);
e1r=round_sat(ar64*65536-tr);e1i=round_sat(ai64*65536-ti);
@(negedge clk);
if(busy)$fatal(1,"BF: intento de caso con DUT ocupado");
a_re=ar;a_im=ai;b_re=br;b_im=bi;w_re=wr;w_im=wi;start=1;
@(posedge clk);#1;
if(!busy||done)$fatal(1,"BF: start no capturado en C0");
@(negedge clk);
a_re=18'sd131071;a_im=-18'sd131072;
b_re=-18'sd131072;b_im=18'sd131071;
w_re=18'sd12345;w_im=-18'sd54321;
start=inject_busy_start;
for(elapsed=1;elapsed<=3;elapsed=elapsed+1)begin
@(posedge clk);#1;
if(elapsed<3&&(!busy||done))
$fatal(1,"BF: latencia incorrecta en C%0d",elapsed);
if(elapsed==3)begin
if(busy||!done)$fatal(1,"BF: resultado debe estar en C3");
if(y0_re!==e0r||y0_im!==e0i||y1_re!==e1r||y1_im!==e1i)
$fatal(1,"BF caso %0d: got (%0d,%0d),(%0d,%0d); esperado (%0d,%0d),(%0d,%0d)",
cases_checked,y0_re,y0_im,y1_re,y1_im,e0r,e0i,e1r,e1i);
end
@(negedge clk);start=0;
end
repeat(3)begin
@(posedge clk);#1;
if(done||busy)$fatal(1,"BF: done largo o start ocupado quedo en cola");
end
cases_checked=cases_checked+1;
end
endtask
integer n;
reg signed[17:0]ra,ia,rb,ib,rw,iw;
initial begin
$dumpfile("fft_butterfly.vcd");$dumpvars(0,tb_fft_butterfly);
reset_dut();
run_case(0,0,0,0,65536,0,0);
run_case(12003,-7001,-23001,11007,65536,0,1);
run_case(12003,-7001,-23001,11007,-65536,0,1);
run_case(12003,-7001,-23001,11007,0,65536,1);
run_case(12003,-7001,-23001,11007,0,-65536,1);
run_case(-32768,32767,23111,-17003,46341,-46341,1);
run_case(1,-1,0,0,0,0,0);
run_case(0,0,1,-1,65536,0,0);
run_case(0,0,-1,1,65536,0,0);
run_case(3,-3,0,0,0,0,0);
run_case(1,-1,1,-1,1,0,0);
run_case(131071,131071,131071,-131072,131071,-131072,1);
run_case(-131072,-131072,-131072,131071,131071,-131072,1);
run_case(131071,-131072,-131072,131071,-131072,131071,1);
for(n=0;n<128;n=n+1)begin
ra=random18();ia=random18();rb=random18();ib=random18();
rw=random18();iw=random18();
run_case(ra,ia,rb,ib,rw,iw,1);
end
@(negedge clk);start=1;a_re=123;b_re=456;w_re=65536;
@(posedge clk);#1;
if(!busy)$fatal(1,"BF: no inicio caso para reset ocupado");
@(negedge clk);start=0;rst_n=0;
#1;
if(busy||done)$fatal(1,"BF: reset ocupado no aborta");
repeat(3)@(negedge clk);
rst_n=1;
repeat(5)begin
@(posedge clk);#1;
if(busy||done)$fatal(1,"BF: resultado viejo reaparecio tras reset");
end
run_case(-32768,0,32767,0,65536,0,1);
$display("PASS: MARIPOSA FFT, %0d casos bit exactos, latencia C3, reset y start ocupado",cases_checked);
$finish;
end
initial begin#1000000;$fatal(1,"BF: timeout global");end
endmodule

previous_begin=0;previous_done=0;
end else if(sample_valid)begin
if(check_cadence&&last_sample_cycle>=0&&cycle_no-last_sample_cycle!=135)
$fatal(1,"PROCESSOR: entrada no mantiene 135 ciclos; intervalo=%0d",cycle_no-last_sample_cycle);
last_sample_cycle=cycle_no;
if(mode==MODE_SUCCESS)begin
input_frame=physical_count/256;
input_index=physical_count%256;
if(input_frame>=frame_limit||input_frame>2)
$fatal(1,"PROCESSOR: mas muestras que los bloques esperados");
actual_input[input_frame][input_index]=sample_signed;
if(input_index==255)build_dft_magnitude(input_frame);
end
physical_count=physical_count+1;
end
#1;
if(rst_n)begin
if(frame_begin&&previous_begin)
$fatal(1,"PROCESSOR: frame_begin dura mas de un ciclo");
if(frame_done&&previous_done)
$fatal(1,"PROCESSOR: frame_done dura mas de un ciclo");
previous_begin=frame_begin;
previous_done=frame_done;
if(fault_latched&&(frame_begin||wave_valid||spectrum_valid||frame_done))
$fatal(1,"PROCESSOR: publico strobes despues de un fault retenido");
if(mode==MODE_OVERLOAD&&frame_done)
$fatal(1,"PROCESSOR: sobrecarga publico un bloque incompleto");
if(mode==MODE_SUCCESS)begin
if(fault_latched)$fatal(1,"PROCESSOR: fallo inesperado %0d",fault_code);
if(frame_begin)begin
if(active_frame>=0)$fatal(1,"PROCESSOR: superposicion de bloques publicados");
if(completed_frames>=frame_limit||frame_seq!==completed_frames)
$fatal(1,"PROCESSOR: secuencia %0d; esperada %0d",frame_seq,completed_frames);
if(physical_count<(completed_frames+1)*256)
$fatal(1,"PROCESSOR: frame_begin antes de capturar 256 muestras");
active_frame=completed_frames;
frame_start_cycle=cycle_no;
wave_count=0;spectrum_count=0;
end
if(active_frame>=0&&frame_seq!==active_frame)
$fatal(1,"PROCESSOR: frame_seq cambio durante procesamiento");
if(wave_valid)begin
if(active_frame<0||wave_count>=256||wave_index!==wave_count)
$fatal(1,"PROCESSOR: indice/orden de traza incorrecto idx=%0d esperado=%0d",wave_index,wave_count);
if(wave_sample!==actual_input[active_frame][wave_count])
$fatal(1,"PROCESSOR: traza frame%0d idx%0d got=%0d esperado=%0d",
active_frame,wave_count,wave_sample,actual_input[active_frame][wave_count]);
wave_count=wave_count+1;
phase_waves=phase_waves+1;
end
if(spectrum_valid)begin
if(active_frame<0||wave_count!=256||spectrum_count>128||
spectrum_index!==spectrum_count)
$fatal(1,"PROCESSOR: indice/orden espectral incorrecto idx=%0d esperado=%0d",spectrum_index,spectrum_count);
if(^spectrum_mag===1'bx)
$fatal(1,"PROCESSOR: magnitud desconocida");
got_magnitude=spectrum_mag;
magnitude_error=abs_real(got_magnitude-ideal_magnitude[active_frame][spectrum_count]);
if(magnitude_error>maximum_magnitude_error)
maximum_magnitude_error=magnitude_error;
if(magnitude_error>MAG_TOLERANCE_LSB)
$fatal(1,"PROCESSOR: frame%0d bin%0d mag=%0d DFTideal=%0.6f error=%0.6f",
active_frame,spectrum_count,got_magnitude,
ideal_magnitude[active_frame][spectrum_count],magnitude_error);
if(spectrum_index==128&&!frame_done)
$fatal(1,"PROCESSOR: bin128 debe coincidir con frame_done");
spectrum_count=spectrum_count+1;
phase_bins=phase_bins+1;
end
if(frame_done)begin
if(active_frame<0||!spectrum_valid||spectrum_index!=128||
spectrum_count!=129||wave_count!=256)
$fatal(1,"PROCESSOR: frame_done sin salida completa");
latency=cycle_no-frame_start_cycle;
if(latency>FRAME_DEADLINE)
$fatal(1,"PROCESSOR: latencia %0d excede %0d ciclos",latency,FRAME_DEADLINE);
if(latency>maximum_frame_cycles)maximum_frame_cycles=latency;
$display("PROCESSOR: frame%0d, 256 muestras, 129 bins, %0d ciclos",active_frame,latency);
completed_frames=completed_frames+1;
active_frame=-1;
end
if(active_frame>=0&&cycle_no-frame_start_cycle>FRAME_DEADLINE)
$fatal(1,"PROCESSOR: bloque excedio plazo sin frame_done");
end
end
end
