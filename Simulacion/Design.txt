module view_selector#(
parameter integer DEBOUNCE_CYCLES=1_485_000
)(
input wire clk,
input wire rst_n,
input wire key_s2,
input wire frame_start,
output reg mode_fft
);
localparam integer COUNT_WIDTH=
(DEBOUNCE_CYCLES>1)?$clog2(DEBOUNCE_CYCLES):1;
localparam[COUNT_WIDTH-1:0]LAST_COUNT=DEBOUNCE_CYCLES-1;
reg key_meta;
reg key_sync;
reg key_stable;
reg[COUNT_WIDTH-1:0]opposite_count;
reg pending_mode;
always@(posedge clk or negedge rst_n)begin
if(!rst_n)begin
key_meta<=1'b0;
key_sync<=1'b0;
key_stable<=1'b0;
opposite_count<={COUNT_WIDTH{1'b0}};
pending_mode<=1'b0;
mode_fft<=1'b0;
end else begin
key_meta<=key_s2;
key_sync<=key_meta;
if(key_sync==key_stable)begin
opposite_count<={COUNT_WIDTH{1'b0}};
end else if(opposite_count==LAST_COUNT)begin
opposite_count<={COUNT_WIDTH{1'b0}};
key_stable<=key_sync;
if(key_sync)
pending_mode<=~pending_mode;
end else begin
opposite_count<=opposite_count+1'b1;
end
if(frame_start)
mode_fft<=pending_mode;
end
end
endmodule

module ad7606_byte_reader#(
parameter integer STARTUP_CYCLES=1_350_000
)(
input wire clk,
input wire rst_n,
input wire adc_busy,
input wire adc_fd,
input wire[7:0]adc_db,
output reg adc_convst,
output reg adc_cs_n,
output reg adc_rd_n,
output reg adc_reset,
output reg signed[15:0]sample_signed,
output reg sample_valid,
output reg fault_latched,
output reg[3:0]fault_code
);
localparam integer STARTUP_WIDTH=
(STARTUP_CYCLES>1)?$clog2(STARTUP_CYCLES):1;
localparam[7:0]PERIOD_LAST=8'd134;
localparam[7:0]BUSY_RISE_LAST=8'd7;
localparam[7:0]BUSY_FALL_LAST=8'd127;
localparam[3:0]MSB_RD_LAST=4'd9;
localparam[3:0]LSB_RD_LAST=4'd1;
typedef enum logic[3:0]{
WAIT_POWER,HOLD_RESET,RESET_GAP,WAIT_BUSY_RISE,WAIT_BUSY_FALL,
READ_SETUP,READ_LOW,READ_GAP,READ_COMMIT,CONVERSION_DONE
}state_t;
state_t state;
reg[STARTUP_WIDTH-1:0]startup_count;
reg[1:0]reset_count;
reg reset_gap_count;
reg[7:0]cycle_count;
reg[3:0]rd_count;
reg second_byte;
reg busy_low_seen;
reg[7:0]msb;
reg signed[15:0]pending_sample;
reg busy_meta,busy_sync;
reg[3:0]pending_fault;
wire running=(state!=WAIT_POWER)&&(state!=HOLD_RESET)&&(state!=RESET_GAP);
wire reading=(state==READ_SETUP)||(state==READ_LOW)||
(state==READ_GAP)||(state==READ_COMMIT);
wire rd_due=(rd_count==(second_byte?LSB_RD_LAST:MSB_RD_LAST));
always@(posedge clk or negedge rst_n)begin
if(!rst_n)begin
busy_meta<=1'b0;
busy_sync<=1'b0;
end else begin
busy_meta<=adc_busy;
busy_sync<=busy_meta;
end
end
always_comb begin
pending_fault=4'd0;
if(reading&&busy_sync)
pending_fault=4'd4;
else if((state==WAIT_BUSY_RISE)&&!busy_sync&&(cycle_count>=BUSY_RISE_LAST))
pending_fault=4'd1;
else if((state==WAIT_BUSY_FALL)&&busy_sync&&(cycle_count>=BUSY_FALL_LAST))
pending_fault=4'd2;
else if(running&&(cycle_count==PERIOD_LAST)&&
((state!=CONVERSION_DONE)||busy_sync))
pending_fault=4'd5;
else if((state==READ_LOW)&&rd_due)begin
case(adc_fd)
1'b1:begin end
default:pending_fault=4'd3;
endcase
end
end
always@(posedge clk or negedge rst_n)begin
if(!rst_n)begin
state<=WAIT_POWER;
startup_count<='0;
reset_count<='0;
reset_gap_count<=1'b0;
cycle_count<='0;
rd_count<='0;
second_byte<=1'b0;
busy_low_seen<=1'b0;
msb<='0;
pending_sample<='0;
adc_convst<=1'b0;
adc_cs_n<=1'b1;
adc_rd_n<=1'b1;
adc_reset<=1'b0;
sample_signed<='0;
sample_valid<=1'b0;
fault_latched<=1'b0;
fault_code<=4'd0;
end else begin
sample_valid<=1'b0;
if(fault_latched||(pending_fault!=4'd0))begin
adc_convst<=1'b0;
adc_cs_n<=1'b1;
adc_rd_n<=1'b1;
adc_reset<=1'b0;
if(!fault_latched)begin
fault_latched<=1'b1;
fault_code<=pending_fault;
end
end else if(!running)begin
case(state)
WAIT_POWER:begin
if(startup_count==STARTUP_CYCLES-1)begin
reset_count<=2'd0;
adc_reset<=1'b1;
state<=HOLD_RESET;
end else startup_count<=startup_count+1'b1;
end
HOLD_RESET:begin
if(reset_count==2'd3)begin
reset_gap_count<=1'b0;
adc_reset<=1'b0;
state<=RESET_GAP;
end else reset_count<=reset_count+1'b1;
end
RESET_GAP:begin
if(reset_gap_count)begin
cycle_count<=8'd0;
adc_convst<=1'b1;
state<=WAIT_BUSY_RISE;
end else reset_gap_count<=1'b1;
end
default:state<=WAIT_POWER;
endcase
end else if(cycle_count==PERIOD_LAST)begin
cycle_count<=8'd0;
adc_convst<=1'b1;
state<=WAIT_BUSY_RISE;
end else begin
cycle_count<=cycle_count+1'b1;
if(cycle_count==8'd1)adc_convst<=1'b0;
case(state)
WAIT_BUSY_RISE:begin
if(busy_sync)begin
busy_low_seen<=1'b0;
state<=WAIT_BUSY_FALL;
end
end
WAIT_BUSY_FALL:begin
if(busy_sync)busy_low_seen<=1'b0;
else if(busy_low_seen)begin
second_byte<=1'b0;
adc_cs_n<=1'b0;
state<=READ_SETUP;
end else busy_low_seen<=1'b1;
end
READ_SETUP,READ_GAP:begin
adc_rd_n<=1'b0;
rd_count<=4'd0;
state<=READ_LOW;
end
READ_LOW:begin
if(rd_due)begin
adc_rd_n<=1'b1;
if(second_byte)begin
pending_sample<=$signed({msb,adc_db});
state<=READ_COMMIT;
end else begin
msb<=adc_db;
second_byte<=1'b1;
state<=READ_GAP;
end
end else rd_count<=rd_count+1'b1;
end
READ_COMMIT:begin
adc_cs_n<=1'b1;
sample_signed<=pending_sample;
sample_valid<=1'b1;
state<=CONVERSION_DONE;
end
default:begin end
endcase
end
end
end
endmodule

module sample_capture(
input wire clk,
input wire rst_n,
input wire sample_valid,
input wire signed[15:0]sample_signed,
output reg block_valid,
output reg block_bank,
output reg[31:0]block_seq,
input wire block_release,
output reg overrun_latched,
input wire rd_en,
input wire rd_bank,
input wire[7:0]rd_addr,
output wire signed[15:0]rd_sample,
output reg rd_valid
);
reg signed[15:0]bank0[0:255];
reg signed[15:0]bank1[0:255];
reg signed[15:0]bank0_read;
reg signed[15:0]bank1_read;
reg write_bank;
reg[7:0]write_addr;
reg[31:0]next_seq;
reg read_bank_q;
wire write_accept=rst_n&&sample_valid&&!overrun_latched;
wire read_accept=rst_n&&rd_en&&block_valid&&
(rd_bank==block_bank);
always@(posedge clk)begin
if(write_accept&&!write_bank)
bank0[write_addr]<=sample_signed;
if(read_accept&&!rd_bank)
bank0_read<=bank0[rd_addr];
end
always@(posedge clk)begin
if(write_accept&&write_bank)
bank1[write_addr]<=sample_signed;
if(read_accept&&rd_bank)
bank1_read<=bank1[rd_addr];
end
assign rd_sample=read_bank_q?bank1_read:bank0_read;
always@(posedge clk or negedge rst_n)begin
if(!rst_n)begin
write_bank<=1'b0;
write_addr<=8'd0;
next_seq<=32'd0;
block_valid<=1'b0;
block_bank<=1'b0;
block_seq<=32'd0;
overrun_latched<=1'b0;
read_bank_q<=1'b0;
rd_valid<=1'b0;
end else begin
rd_valid<=read_accept;
if(read_accept)
read_bank_q<=rd_bank;
if(!overrun_latched)begin
if(block_release&&block_valid)
block_valid<=1'b0;
if(sample_valid)begin
if(write_addr==8'd255)begin
if(block_valid&&!block_release)begin
overrun_latched<=1'b1;
end else begin
block_valid<=1'b1;
block_bank<=write_bank;
block_seq<=next_seq;
next_seq<=next_seq+32'd1;
write_bank<=~write_bank;
write_addr<=8'd0;
end
end else begin
write_addr<=write_addr+8'd1;
end
end
end
end
end
endmodule

module fft_butterfly(
input logic clk,
input logic rst_n,
input logic start,
input logic signed[17:0]a_re,
input logic signed[17:0]a_im,
input logic signed[17:0]b_re,
input logic signed[17:0]b_im,
input logic signed[17:0]w_re,
input logic signed[17:0]w_im,
output logic busy,
output logic done,
output logic signed[17:0]y0_re,
output logic signed[17:0]y0_im,
output logic signed[17:0]y1_re,
output logic signed[17:0]y1_im
);
typedef enum logic[1:0]{IDLE,MULTIPLY,SUM_PRODUCTS,OUTPUTS}state_t;
state_t state;
logic signed[17:0]a_re_reg,a_im_reg;
logic signed[17:0]b_re_reg,b_im_reg;
logic signed[17:0]w_re_reg,w_im_reg;
logic signed[35:0]product_rr,product_ii,product_ri,product_ir;
logic signed[36:0]rotated_re,rotated_im;
wire signed[37:0]a_re_scaled=
$signed({{20{a_re_reg[17]}},a_re_reg})<<<16;
wire signed[37:0]a_im_scaled=
$signed({{20{a_im_reg[17]}},a_im_reg})<<<16;
wire signed[37:0]rotated_re_wide={rotated_re[36],rotated_re};
wire signed[37:0]rotated_im_wide={rotated_im[36],rotated_im};
function automatic logic signed[17:0]round_and_saturate(
input logic signed[37:0]value
);
logic signed[38:0]extended_value;
logic[38:0]magnitude;
logic[38:0]rounded_magnitude;
begin
extended_value={value[37],value};
if(value[37])
magnitude=$unsigned(-extended_value);
else
magnitude=$unsigned(extended_value);
rounded_magnitude=(magnitude+39'd65536)>>17;
if(value[37])begin
if(rounded_magnitude>=39'd131072)
round_and_saturate=18'sh20000;
else
round_and_saturate=-$signed(rounded_magnitude[17:0]);
end else begin
if(rounded_magnitude>39'd131071)
round_and_saturate=18'sh1ffff;
else
round_and_saturate=$signed(rounded_magnitude[17:0]);
end
end
endfunction
always_ff@(posedge clk or negedge rst_n)begin
if(!rst_n)begin
state<=IDLE;
busy<=1'b0;
done<=1'b0;
a_re_reg<='0;
a_im_reg<='0;
b_re_reg<='0;
b_im_reg<='0;
w_re_reg<='0;
w_im_reg<='0;
product_rr<='0;
product_ii<='0;
product_ri<='0;
product_ir<='0;
rotated_re<='0;
rotated_im<='0;
y0_re<='0;
y0_im<='0;
y1_re<='0;
y1_im<='0;
end else begin
done<=1'b0;
case(state)
IDLE:begin
if(start)begin
a_re_reg<=a_re;
a_im_reg<=a_im;
b_re_reg<=b_re;
b_im_reg<=b_im;
w_re_reg<=w_re;
w_im_reg<=w_im;
busy<=1'b1;
state<=MULTIPLY;
end
end
MULTIPLY:begin
product_rr<=b_re_reg*w_re_reg;
product_ii<=b_im_reg*w_im_reg;
product_ri<=b_re_reg*w_im_reg;
product_ir<=b_im_reg*w_re_reg;
state<=SUM_PRODUCTS;
end
SUM_PRODUCTS:begin
rotated_re<=$signed({product_rr[35],product_rr})-
$signed({product_ii[35],product_ii});
rotated_im<=$signed({product_ri[35],product_ri})+
$signed({product_ir[35],product_ir});
state<=OUTPUTS;
end
OUTPUTS:begin
y0_re<=round_and_saturate(a_re_scaled+rotated_re_wide);
y0_im<=round_and_saturate(a_im_scaled+rotated_im_wide);
y1_re<=round_and_saturate(a_re_scaled-rotated_re_wide);
y1_im<=round_and_saturate(a_im_scaled-rotated_im_wide);
busy<=1'b0;
done<=1'b1;
state<=IDLE;
end
endcase
end
end
endmodule

module fft256_core(
input logic clk,
input logic rst_n,
input logic in_valid,
input logic signed[15:0]in_sample,
output wire in_ready,
input logic start,
output logic frame_loaded,
output logic busy,
output logic done,
output logic results_ready,
input logic rd_en,
input logic[7:0]rd_addr,
output wire signed[17:0]rd_re,
output wire signed[17:0]rd_im,
output logic rd_valid,
output logic fault_latched,
output logic[3:0]fault_code
);
typedef enum logic[2:0]{
IDLE,READ_A,READ_B,BF_START,BF_WAIT,WRITE_A,WRITE_B
}state_t;
state_t state;
logic[7:0]load_index;
logic[2:0]stage_index;
logic[7:0]group_base;
logic[6:0]pair_index;
wire[8:0]half_span=9'd1<<stage_index;
wire[8:0]group_span=9'd2<<stage_index;
wire[8:0]group_end={1'b0,group_base}+group_span;
wire[8:0]address_a_wide={1'b0,group_base}+{2'b00,pair_index};
wire[8:0]address_b_wide=address_a_wide+half_span;
wire[7:0]address_a=address_a_wide[7:0];
wire[7:0]address_b=address_b_wide[7:0];
wire[6:0]twiddle_addr=pair_index<<(3'd7-stage_index);
wire input_accept=in_valid&&in_ready;
wire result_read_accept=rd_en&&results_ready&&!busy&&
!input_accept&&!start&&!fault_latched;
wire fault_pending=start&&(busy||!frame_loaded);
wire[3:0]pending_fault_code=busy?4'd2:4'd1;
logic[35:0]memory[0:255];
logic[35:0]memory_q;
logic memory_en,memory_we;
logic[7:0]memory_addr;
logic[35:0]memory_d;
logic signed[17:0]operand_a_re,operand_a_im;
wire signed[17:0]twiddle_re,twiddle_im;
wire signed[17:0]result_a_re,result_a_im,result_b_re,result_b_im;
wire butterfly_busy,butterfly_done;
wire butterfly_start=(state==BF_START)&&!fault_latched&&!fault_pending;
wire twiddle_en=(state==READ_A)&&!fault_latched&&!fault_pending;
assign in_ready=rst_n&&!busy&&!frame_loaded&&!fault_latched;
assign rd_re=$signed(memory_q[35:18]);
assign rd_im=$signed(memory_q[17:0]);
function automatic logic[7:0]reverse_bits(input logic[7:0]value);
integer bit_index;
begin
for(bit_index=0;bit_index<8;bit_index=bit_index+1)
reverse_bits[bit_index]=value[7-bit_index];
end
endfunction
fft_twiddle_rom twiddles(
.clk(clk),.en(twiddle_en),.addr(twiddle_addr),
.w_re(twiddle_re),.w_im(twiddle_im)
);
fft_butterfly butterfly(
.clk(clk),.rst_n(rst_n),.start(butterfly_start),
.a_re(operand_a_re),.a_im(operand_a_im),
.b_re($signed(memory_q[35:18])),.b_im($signed(memory_q[17:0])),
.w_re(twiddle_re),.w_im(twiddle_im),
.busy(butterfly_busy),.done(butterfly_done),
.y0_re(result_a_re),.y0_im(result_a_im),
.y1_re(result_b_re),.y1_im(result_b_im)
);
always_comb begin
memory_en=1'b0;
memory_we=1'b0;
memory_addr='0;
memory_d='0;
if(rst_n&&!fault_latched&&!fault_pending)begin
case(state)
IDLE:begin
if(input_accept)begin
memory_en=1'b1;
memory_we=1'b1;
memory_addr=reverse_bits(load_index);
memory_d={{2{in_sample[15]}},in_sample,18'd0};
end else if(result_read_accept)begin
memory_en=1'b1;
memory_addr=rd_addr;
end
end
READ_A:begin
memory_en=1'b1;
memory_addr=address_a;
end
READ_B:begin
memory_en=1'b1;
memory_addr=address_b;
end
WRITE_A:begin
memory_en=1'b1;
memory_we=1'b1;
memory_addr=address_a;
memory_d={result_a_re,result_a_im};
end
WRITE_B:begin
memory_en=1'b1;
memory_we=1'b1;
memory_addr=address_b;
memory_d={result_b_re,result_b_im};
end
default:begin end
endcase
end
end
always_ff@(posedge clk)begin
if(memory_en)begin
if(memory_we)
memory[memory_addr]<=memory_d;
else
memory_q<=memory[memory_addr];
end
end
always_ff@(posedge clk or negedge rst_n)begin
if(!rst_n)begin
state<=IDLE;
load_index<='0;
stage_index<='0;
group_base<='0;
pair_index<='0;
operand_a_re<='0;
operand_a_im<='0;
frame_loaded<=1'b0;
busy<=1'b0;
done<=1'b0;
results_ready<=1'b0;
rd_valid<=1'b0;
fault_latched<=1'b0;
fault_code<=4'd0;
end else begin
done<=1'b0;
rd_valid<=1'b0;
if(fault_latched||fault_pending)begin
state<=IDLE;
busy<=1'b0;
frame_loaded<=1'b0;
results_ready<=1'b0;
if(!fault_latched)begin
fault_latched<=1'b1;
fault_code<=pending_fault_code;
end
end else begin
case(state)
IDLE:begin
rd_valid<=result_read_accept;
if(start)begin
busy<=1'b1;
frame_loaded<=1'b0;
results_ready<=1'b0;
load_index<='0;
stage_index<='0;
group_base<='0;
pair_index<='0;
state<=READ_A;
end else if(input_accept)begin
results_ready<=1'b0;
if(load_index==8'd255)begin
frame_loaded<=1'b1;
end else begin
load_index<=load_index+1'b1;
end
end
end
READ_A:begin
state<=READ_B;
end
READ_B:begin
operand_a_re<=$signed(memory_q[35:18]);
operand_a_im<=$signed(memory_q[17:0]);
state<=BF_START;
end
BF_START:begin
state<=BF_WAIT;
end
BF_WAIT:begin
if(butterfly_done)
state<=WRITE_A;
end
WRITE_A:begin
state<=WRITE_B;
end
WRITE_B:begin
if({2'b00,pair_index}==half_span-9'd1)begin
pair_index<='0;
if(group_end==9'd256)begin
group_base<='0;
if(stage_index==3'd7)begin
busy<=1'b0;
done<=1'b1;
results_ready<=1'b1;
state<=IDLE;
end else begin
stage_index<=stage_index+1'b1;
state<=READ_A;
end
end else begin
group_base<=group_end[7:0];
state<=READ_A;
end
end else begin
pair_index<=pair_index+1'b1;
state<=READ_A;
end
end
default:begin end
endcase
end
end
end
endmodule

module fft_magnitude(
input logic clk,
input logic rst_n,
input logic start,
input logic signed[17:0]in_re,
input logic signed[17:0]in_im,
output logic busy,
output logic done,
output logic[17:0]magnitude
);
typedef enum logic[1:0]{IDLE,SQUARE,INITIALIZE,ROOT_ITERATION}state_t;
state_t state;
logic signed[17:0]re_reg,im_reg;
logic signed[35:0]square_re,square_im;
wire[36:0]square_sum={1'b0,$unsigned(square_re)}+
{1'b0,$unsigned(square_im)};
logic[35:0]radicand_shift;
logic[37:0]remainder;
logic[17:0]partial_root;
logic[4:0]iteration;
wire[37:0]expanded_remainder=
(remainder<<2)|{36'd0,radicand_shift[35:34]};
wire[37:0]trial_divisor=({20'd0,partial_root}<<2)|38'd1;
wire take_bit=expanded_remainder>=trial_divisor;
wire[37:0]next_remainder=take_bit
?expanded_remainder-trial_divisor:expanded_remainder;
wire[17:0]next_root={partial_root[16:0],take_bit};
always_ff@(posedge clk or negedge rst_n)begin
if(!rst_n)begin
state<=IDLE;
re_reg<='0;
im_reg<='0;
square_re<='0;
square_im<='0;
radicand_shift<='0;
remainder<='0;
partial_root<='0;
iteration<='0;
busy<=1'b0;
done<=1'b0;
magnitude<='0;
end else begin
done<=1'b0;
case(state)
IDLE:begin
if(start)begin
re_reg<=in_re;
im_reg<=in_im;
busy<=1'b1;
state<=SQUARE;
end
end
SQUARE:begin
square_re<=re_reg*re_reg;
square_im<=im_reg*im_reg;
state<=INITIALIZE;
end
INITIALIZE:begin
radicand_shift<=square_sum[35:0];
remainder<='0;
partial_root<='0;
iteration<='0;
state<=ROOT_ITERATION;
end
ROOT_ITERATION:begin
radicand_shift<={radicand_shift[33:0],2'b00};
remainder<=next_remainder;
partial_root<=next_root;
if(iteration==5'd17)begin
magnitude<=next_root;
busy<=1'b0;
done<=1'b1;
state<=IDLE;
end else begin
iteration<=iteration+1'b1;
end
end
endcase
end
end
endmodule

end
end

FAILED:begin

end

active<=1'b0;

working_seq<=frame_seq;

end
end

request_meta<=1'b0;

if(!pix_rst_n)begin
front_bank<=1'b0;
front_valid<=1'b0;
front_seq<=32'd0;
ack_toggle<=1'b0;
end else if(frame_start&&(request_sync!=ack_toggle))begin
front_bank<=published_bank;
front_seq<=published_seq;

end
end
reg[31:0]wave_bank0[0:255];
reg[31:0]wave_bank1[0:255];
reg[17:0]spectrum_bank0[0:255];
reg[17:0]spectrum_bank1[0:255];

always@(posedge wr_clk)begin
if(wave_write&&back_bank)

module tmds_encoder(
input logic clk,
input logic rst_n,
input logic de,
input logic[7:0]data,
input logic c0,
input logic c1,
output logic[9:0]symbol
);
localparam logic[9:0]CONTROL_00=10'b1101010100;
logic signed[5:0]disparity,disparity_next;
logic[9:0]symbol_next;
logic[8:0]q_m;
logic[3:0]ones_data,ones_q;
logic use_xnor;
logic signed[5:0]payload_balance;
integer bit_index;
always_comb begin
ones_data=4'd0;
for(bit_index=0;bit_index<8;bit_index=bit_index+1)
ones_data=ones_data+{3'b000,data[bit_index]};
use_xnor=(ones_data>4)||((ones_data==4)&&!data[0]);
q_m='0;
q_m[0]=data[0];
q_m[8]=!use_xnor;
for(bit_index=1;bit_index<8;bit_index=bit_index+1)begin
if(use_xnor)
q_m[bit_index]=~(q_m[bit_index-1]^data[bit_index]);
else
q_m[bit_index]=q_m[bit_index-1]^data[bit_index];
end
ones_q=4'd0;
for(bit_index=0;bit_index<8;bit_index=bit_index+1)
ones_q=ones_q+{3'b000,q_m[bit_index]};
payload_balance=$signed({1'b0,ones_q,1'b0})-6'sd8;
symbol_next=CONTROL_00;
disparity_next=6'sd0;
if(!de)begin
case({c1,c0})
2'b00:symbol_next=10'b1101010100;
2'b01:symbol_next=10'b0010101011;
2'b10:symbol_next=10'b0101010100;
2'b11:symbol_next=10'b1010101011;
default:symbol_next=CONTROL_00;
endcase
end else if((disparity==0)||(payload_balance==0))begin
symbol_next={~q_m[8],q_m[8],q_m[8]?q_m[7:0]:~q_m[7:0]};
if(q_m[8])
disparity_next=disparity+payload_balance;
else
disparity_next=disparity-payload_balance;
end else if(((disparity>0)&&(payload_balance>0))||
((disparity<0)&&(payload_balance<0)))begin
symbol_next={1'b1,q_m[8],~q_m[7:0]};
disparity_next=disparity-payload_balance;
if(q_m[8])disparity_next=disparity_next+6'sd2;
end else begin
symbol_next={1'b0,q_m[8],q_m[7:0]};
disparity_next=disparity+payload_balance;
if(!q_m[8])disparity_next=disparity_next-6'sd2;
end
end
always_ff@(posedge clk or negedge rst_n)begin
if(!rst_n)begin
symbol<=CONTROL_00;
disparity<=6'sd0;
end else begin
symbol<=symbol_next;
disparity<=disparity_next;
end
end
endmodule

);
localparam[3:0]NONE=0,PROJECT=1,TITLE=2,AUTHORS=3,
TIME_AXIS=5,FREQ_AXIS=6,ZERO=7,
TIME_1=8,TIME_2=9,TIME_3=10,TIME_4=11,
FREQ_1=12,FREQ_2=13,FREQ_3=14,FREQ_4=15;
function automatic[7:0]character_at(input[3:0]line,input[5:0]column);
reg[511:0]line_data;
begin
case(line)
PROJECT:line_data={"Proyecto final ELDI",360'd0};
TITLE:line_data={"Analizador FFT",400'd0};

TIME_3:line_data={"0,96",480'd0};

FREQ_3:line_data={"75",496'd0};

glyph_row_0<=title_scale?text_dy[4:2]:text_dy[3:1];

logo_1<=logo_0?logo_level:2'd0;

tick_2<=tick_1;

else if(text_2)rgb_out<=24'hacb8c8;
else if(tick_2)rgb_out<=24'h758498;
else rgb_out<=rgb_2;

end
